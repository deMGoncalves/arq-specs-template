### 4. Test-First (TDD)
```typescript
describe('{{COMPONENT_NAME}}', () => {
  describe('{{USE_CASE}}', () => {
    it('should {{comportamento esperado}}', () => {
      // Given (Arrange)
      const [precondições]

      // When (Act)
      const result = [ação]

      // Then (Assert)
      expect(result).toEqual([esperado])
    });

    it('should throw {{ERROR}} when {{condição}}', () => {
      // Given
      const [condições inválidas]

      // When/Then
      expect(() => [ação]).toThrow({{ERROR}})
    });
  });
});
```

### 5. Mapeamento BDD

```gherkin

```
